src = open(os.path.join(HERE, 'mutate_m.py'), encoding='utf-8').read()

exec(src.split('\n\ndef sha(path):')[0].replace('WT = sys.argv[1]', "WT = 'C:/Users/jonathan/Desktop/KCV-of1102'")
