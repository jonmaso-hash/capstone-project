def dedent(text):
    """The same lines, one indent (4 spaces) shallower: _one_self_reported's body moved out of a loop."""
    lines = text.split('\n')
    return '\n'.join(l[4:] if l.startswith('    ') else l for l in lines)

MUTATIONS = []
for name, rel, old, new in ns['MUTATIONS']:
    key = name.split()[0]
    if key in RETARGET:
        old, new = RETARGET[key]
    elif rel == FF and source.count(old) != 1 and source.count(dedent(old)) == 1:
        old, new = dedent(old), dedent(new)
    MUTATIONS.append((name, rel, old, new))
