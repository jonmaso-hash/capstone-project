def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()


originals = {k: open(p, 'rb').read() for k, p in FILES.items()}
before = {k: sha(p) for k, p in FILES.items()}
for mid, which, desc, old, new in M:
    if ONLY and mid not in ONLY:
        continue
    path, original = FILES[which], originals[which]
    text = original.decode('utf-8')
    if '\r\n' in text:                         # the working copy's line endings (core.autocrlf)
        old, new = old.replace('\n', '\r\n'), new.replace('\n', '\r\n')
    count = text.count(old)
    if count != 1:
        print(f'{mid} NOT-APPLIED anchor count {count} | {desc}', flush=True)
        continue
    open(path, 'wb').write(text.replace(old, new).encode('utf-8'))
    try:
        run = subprocess.run(['sh', T, WT] + LABELS, capture_output=True, text=True, encoding='utf-8', errors='replace')
        out = run.stdout + run.stderr
    finally:
        open(path, 'wb').write(original)
    restored = all(sha(p) == before[k] for k, p in FILES.items())
    summary = (re.findall(r'^(?:OK.*|FAILED \(.*\))$', out, re.M) or ['?'])[-1]
    ran = (re.findall(r'^Ran \d+ tests.*$', out, re.M) or ['?'])[-1]
    fails = sorted({m for m in re.findall(r'^FAIL: (\w+)', out, re.M)})
    errors = sorted({m for m in re.findall(r'^ERROR: (\w+)', out, re.M)})
    verdict = 'CAUGHT' if fails else ('CRASH-ONLY' if errors else 'SURVIVED')
    if ran == '?':
        verdict = 'NO-RUN'
    print(f'{mid} {verdict} {ran} {summary} restored={restored} | {desc}', flush=True)
    print(f'    assertion failures in: {", ".join(fails) or "-"}', flush=True)
    print(f'    errors in: {", ".join(errors) or "-"}', flush=True)
    for case in sorted(set(re.findall(r'^FAIL: \w+ \([^)]*\) \((.*)\)$', out, re.M)))[:6]:
        print(f'    failing sub-case: {case[:200]}', flush=True)
    if not restored:
        sys.exit('STOP: restore hash mismatch after ' + mid)
