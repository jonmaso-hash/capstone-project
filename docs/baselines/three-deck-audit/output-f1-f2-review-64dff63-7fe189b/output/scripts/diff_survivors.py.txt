"""Differential search for the surviving mutants (review of 64dff63): generated heading-list sentences through the full
guard order with memo 104's payload, at the committed guards and at each surviving mutant applied in memory to the
committed source (anchors read from mutate_f12.py). Any output difference is a counterexample to "equivalent".
The guard file on disk is not touched. No DB, no model call. Deterministic (seeded).
Usage: diff_survivors.py <mutate_f12.py> <out json> [n_random] [survivor ...]"""
import importlib.util, json, random, subprocess, sys
from zelda_api.claim_context import guard_time_wording
from zelda_api import funding_fidelity as fund
from zelda_api import tests_output_fidelity as t

HEAD = subprocess.run(['git', 'rev-parse', 'HEAD'], capture_output=True, text=True, check=True).stdout.strip()
src = subprocess.run(['git', 'show', 'HEAD:zelda_api/basis_fidelity.py'], capture_output=True, text=True, encoding='utf-8',
                     check=True).stdout


def module(code, label):
    mod = importlib.util.module_from_spec(importlib.util.spec_from_loader(f'zelda_api.basis_{label}', loader=None))
    mod.__package__ = 'zelda_api'
    exec(compile(code, f'{label}:basis_fidelity.py', 'exec'), mod.__dict__)
    return mod


defs = open(sys.argv[1], encoding='utf-8').read()
scope = {}
exec(defs[defs.index('M = ['):defs.index('\n]\n') + 3], scope)
SURVIVORS = sys.argv[4:] or ['H2', 'H3', 'H4', 'H6', 'O3', 'P3', 'S1']
committed = module(src, 'committed')
mutants = {}
for mid, which, desc, old, new in scope['M']:
    if mid in SURVIVORS:
        assert which == 'basis' and src.count(old) == 1, mid
        mutants[mid] = module(src.replace(old, new), mid)

payload = t.memo104_payload()
items, notes = t.items_from(payload), t.notes_from(payload)
allowed = {i.ref: i.time_wording_allowed for i in items if i.dated}
ffacts = fund.funding_facts(items, 'ManyChat')
bfacts = {label: mod.basis_facts(items, notes) for label, mod in [('committed', committed)] + list(mutants.items())}

DASH_SUBJECTS = [
    "The company's self-reported traction figures", "The company's unverified traction figures",
    "The self-reported figures", "The company's traction figures",
    "The only indicators of commercial activity are the company's self-reported traction figures",
    "The figures", "Pitch Deck Hunt's traction figures", "An analyst describes the company's traction figures",
    "According to an analyst, the company's self-reported traction figures",
    "If the company's self-reported traction figures", "The company's self-reported traction figures may be",
    "Possibly the company's self-reported traction figures", "Not the company's self-reported traction figures",
    "The company's own self-reported traction figures", "The deck's self-reported traction figures",
    "The company's allegedly self-reported traction figures", "The company's self-reported, unverified traction figures"]
COMMA_SUBJECTS = ["The company states", "The deck states", "Pitch Deck Hunt states", "The company does not state",
                  "If the company states", "The company reportedly states", "An analyst states", "The company never states",
                  "The company may state", "Management states"]
VALUES = {'S1': ['140,000+', '140,000', '140 000+', '140,000++', '~140,000+', '150,000+', '140,000+/month'],
          'S2': ['500M', '500M+', '250M', '$500M', '500M/month', '~500M'],
          'S3': ['500M', '500M+', '250M', '500M/month', '500M+/month']}
NOUNS = {'S1': ['bots', 'messages', 'bots/month', 'bots with no messages', 'bots with no explicit interval',
                'bots (unaudited)', 'bots or more'],
         'S2': ['messages', 'messages with no explicit interval', 'bots', 'messages/week', 'messages (unaudited)',
                'messages with no audit', 'messages with no', 'messages with explicit interval'],
         'S3': ['messages/month', 'messages', 'messages/week', 'messages/month (unaudited)', 'messages/bots', 'month messages']}
HEADS = [" — contextualised under a slide heading 'Traction by Apr-16'",
         " — presented under the 'Traction by Apr-16' slide heading"]
ENDS = ['.', ', but neither is verified.', ', but these figures are undated [C2, C3, C4].']


def figure(ref, rnd):
    if rnd.random() < 0.5:                      # the stated form, so the mutants' paths are reached
        return {'S1': '140,000+ bots', 'S2': '500M messages', 'S3': '500M messages/month'}[ref]
    return f'{rnd.choice(VALUES[ref])} {rnd.choice(NOUNS[ref])}'


def lists(rnd):
    shape = rnd.choice(['s1|s2s3', 's1|s2s3p', 's3|s1s2', 's1|s2|s3', 's1s2|s3', 's1|s2s3x'])
    j = rnd.choice([' and ', ' or ', ', '])
    if shape.startswith('s1|s2s3'):
        body = f"{figure('S1', rnd)} [S1] and {figure('S2', rnd)}{j}{figure('S3', rnd)}"
        if shape.endswith('p'):
            body += ' (see consistency note below)'
        if shape.endswith('x'):
            body += rnd.choice([' figures', ' daily', ''])
        return body + rnd.choice([' [S2, S3]', ' [S2, S3]', ' [S3, S2]', ' [S2, S3, S4]', ' [S2]'])
    if shape == 's3|s1s2':
        return f"{figure('S3', rnd)} [S3] and {figure('S1', rnd)}{j}{figure('S2', rnd)} [S1, S2]"
    if shape == 's1|s2|s3':
        return f"{figure('S1', rnd)} [S1], {figure('S2', rnd)} [S2]{j}{figure('S3', rnd)} [S3]"
    return f"{figure('S1', rnd)}{j}{figure('S2', rnd)} [S1, S2] and {figure('S3', rnd)} [S3]"


def sentence(rnd):
    body = lists(rnd)
    if rnd.random() < 0.6:
        return f"{rnd.choice(DASH_SUBJECTS)} — {body}{rnd.choice(HEADS)}{rnd.choice(ENDS)}"
    return f"{rnd.choice(COMMA_SUBJECTS)} {body}, contextualised under a slide heading 'Traction by Apr-16'{rnd.choice(ENDS)}"


def guard(text, label, mod):
    out, _ = guard_time_wording(text, allowed)
    out, _ = fund.guard_funding_fidelity(out, ffacts)
    return mod.guard_basis_wording(out, bfacts[label])


rnd = random.Random(64)
n = int(sys.argv[3]) if len(sys.argv) > 3 else 6000
seen, cases = set(), []
while len(cases) < n:
    s = sentence(rnd)
    if s not in seen:
        seen.add(s)
        cases.append(s)
count = {mid: 0 for mid in mutants}
examples = {mid: [] for mid in mutants}
rewritten = 0
for s in cases:
    base = guard(s, 'committed', committed)
    rewritten += 'basis_heading_scope' in base[1]
    for mid, mod in mutants.items():
        got = guard(s, mid, mod)
        if got != base:
            count[mid] += 1
            if len(examples[mid]) < 200:
                examples[mid].append({'input': s, 'committed': base, 'mutant': got})
print('head', HEAD, '| sentences', len(cases), '| rewritten by committed', rewritten)
for mid in mutants:
    print(f'{mid}: {count[mid]} differing sentences')
    for ex in examples[mid][:2]:
        print('   in :', ex['input'])
        print('   cmt:', 'kept' if ex['committed'][0] == ex['input'] else ex['committed'])
        print('   mut:', ex['mutant'][1], ex['mutant'][0][:200])
json.dump({'head': HEAD, 'seed': 64, 'n': len(cases), 'rewritten_by_committed': rewritten, 'differing': count,
           'examples': examples}, open(sys.argv[2], 'w', encoding='utf-8', newline='\n'), indent=1, ensure_ascii=False)
