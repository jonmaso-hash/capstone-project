import sys
p = sys.argv[1]
t = open(p, encoding='utf-8', newline='').read()
nl = '\r\n' if '\r\n' in t else '\n'
t = t.replace('\r\n', '\n')
# Review of 3e0fa33 removed the sole-pair fallback: each sentence below is now tied to the note by its noun.
R = [
("        text, changes = self.guard('If the cumulative total is shown as monthly, the claim is inflated.', cumulative=True)",
 "        text, changes = self.guard('If the cumulative messages total is shown as monthly, the claim is inflated.',\n"
 "                                   cumulative=True)"),
("    def test_no_note_no_rule(self):\n"
 "        text = 'If the cumulative figure is presented as monthly, the claim is materially overstated.'",
 "    def test_no_note_no_rule(self):\n"
 "        text = 'If the cumulative messages figure is presented as monthly, the claim is materially overstated.'"),
("        once, _ = self.guard('If the cumulative figure is presented as monthly, the claim is materially overstated.')\n"
 "        self.assertEqual(self.guard(once), (once, []))",
 "        once, changes = self.guard('If the cumulative messages figure is presented as monthly, the claim is materially '\n"
 "                                   'overstated.')\n"
 "        self.assertEqual(changes, ['basis_overstatement'])\n"
 "        self.assertEqual(self.guard(once), (once, []))"),
("        text, _ = self.guard('1. If the cumulative figure is presented as monthly, the claim is overstated.')",
 "        text, _ = self.guard('1. If the cumulative messages figure is presented as monthly, the claim is overstated.')"),
]
for a, b in R:
    assert t.count(a) == 1, a[:70]
    t = t.replace(a, b)
open(p, 'w', encoding='utf-8', newline='').write(t.replace('\n', nl))
print('ok')
