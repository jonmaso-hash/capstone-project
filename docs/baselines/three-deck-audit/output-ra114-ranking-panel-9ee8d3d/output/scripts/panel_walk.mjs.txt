// RA-114 panel walk (evidence script). Headless Chrome with a throwaway profile, driven over the DevTools protocol,
// against verify_panel_server.py on 127.0.0.1 (throwaway SQLite, network blocked, extraction recorded, no model).
// Signs in by setting the seeded throwaway session cookie; no password is typed. Writes walk.json and PNGs to <out>.
// Usage: node panel_walk.mjs <out_dir> <port> <chrome.exe>
import { spawn } from 'node:child_process';
import { mkdtempSync, readFileSync, writeFileSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

const [out, port, chromePath] = process.argv.slice(2);
const seed = JSON.parse(readFileSync(join(out, 'seed.json'), 'utf8'));
const base = `http://127.0.0.1:${port}`;
const profile = mkdtempSync(join(tmpdir(), 'ra114-chrome-'));
const debugPort = 9333;
const chrome = spawn(chromePath, ['--headless=new', `--remote-debugging-port=${debugPort}`, `--user-data-dir=${profile}`,
  '--no-first-run', '--no-default-browser-check', '--window-size=1400,1000', 'about:blank'], { stdio: 'ignore' });
const sleep = ms => new Promise(r => setTimeout(r, ms));

let target;
for (let i = 0; i < 50 && !target; i++) {
  await sleep(200);
  try { target = (await (await fetch(`http://127.0.0.1:${debugPort}/json`)).json()).find(t => t.type === 'page'); } catch {}
}
const ws = new WebSocket(target.webSocketDebuggerUrl);
await new Promise(r => ws.addEventListener('open', r));
let nextId = 0; const pending = new Map(); const consoleErrors = [];
ws.addEventListener('message', ev => {
  const msg = JSON.parse(ev.data);
  if (msg.id && pending.has(msg.id)) { pending.get(msg.id)(msg); pending.delete(msg.id); }
  if (msg.method === 'Runtime.exceptionThrown') consoleErrors.push(msg.params.exceptionDetails.exception?.description || msg.params.exceptionDetails.text);
  if (msg.method === 'Runtime.consoleAPICalled' && msg.params.type === 'error') consoleErrors.push(msg.params.args.map(a => a.value ?? a.description).join(' '));
});
const send = (method, params = {}) => new Promise(r => { const id = ++nextId; pending.set(id, r); ws.send(JSON.stringify({ id, method, params })); });
const evaluate = async expr => {
  const r = await send('Runtime.evaluate', { expression: expr, awaitPromise: true, returnByValue: true });
  if (r.result.exceptionDetails) throw new Error(r.result.exceptionDetails.exception?.description || 'eval error');
  return r.result.result.value;
};
const shot = async name => {
  const r = await send('Page.captureScreenshot', { format: 'png' });
  writeFileSync(join(out, name), Buffer.from(r.result.data, 'base64'));
};
const requests = () => { try { return readFileSync(join(out, 'ask_requests.jsonl'), 'utf8').trim().split('\n').filter(Boolean).map(JSON.parse); } catch { return []; } };
await send('Page.enable'); await send('Runtime.enable'); await send('Network.enable');

async function signIn(username) {
  await send('Network.clearBrowserCookies');
  await send('Network.setCookie', { name: seed.cookie_name, value: seed.sessions[username], domain: '127.0.0.1', path: '/' });
  await send('Page.navigate', { url: `${base}/` });
  for (let i = 0; i < 50; i++) { await sleep(200); if (await evaluate(`document.readyState === 'complete' && location.href.startsWith('${base}')`)) break; }
  await evaluate(`bootstrap.Offcanvas.getOrCreateInstance(document.getElementById('aiAgentSidebar')).show(), true`);
  await sleep(600);
}
const state = () => evaluate(`({
  contextShown: getComputedStyle(document.getElementById('zelda-ask-document')).display !== 'none',
  contextName: document.getElementById('zelda-ask-document-name').textContent,
  placeholder: document.getElementById('zeldaAskQuery').placeholder,
  resultsShown: document.getElementById('ask-results-card').style.display === 'block',
  results: [...document.querySelectorAll('#ask-results-list .zelda-card')].map(c => c.innerText.slice(0, 160)),
  lastLog: [...document.querySelectorAll('#agent-response-log .log-entry')].slice(-2).map(e => e.textContent.replace(/^\\[[^\\]]*\\]\\s*/, '')),
})`);
async function ask(question) {
  const before = requests().length;
  await evaluate(`(() => { document.querySelector('.zelda-tab[data-tab="ask"]').click();
    document.getElementById('zeldaAskQuery').value = ${JSON.stringify(question)};
    document.getElementById('zelda-ask-form').requestSubmit(); return true; })()`);
  for (let i = 0; i < 50 && requests().length === before; i++) await sleep(200);
  await sleep(500);
  const sent = requests().slice(before);
  return { question, sent: sent.map(r => ({ body: r.body, status: r.status, evidence_pages: (r.response.evidence || []).map(e => e.page), no_relevant_evidence: r.response.no_relevant_evidence, response: r.response.response || r.response.message })), panel: await state() };
}

const walk = { base, document_id: seed.document_id, steps: [] };
const step = (name, data) => { walk.steps.push({ name, ...data }); console.log(name, JSON.stringify(data).slice(0, 300)); };

// Owner.
await signIn('verify_owner');
step('positive control: page loaded signed in, panel present', await evaluate(`({ url: location.href, title: document.title,
  askForm: !!document.getElementById('zelda-ask-form'), sidebarShown: document.getElementById('aiAgentSidebar').classList.contains('show'),
  contextShownAtLoad: getComputedStyle(document.getElementById('zelda-ask-document')).display !== 'none' })`));
await shot('01_owner_panel_loaded.png');
step('marketplace question, no document chosen', await ask('Series A founders in messaging'));
await evaluate(`document.querySelector('.zelda-tab[data-tab="library"]').click(), true`);
for (let i = 0; i < 50; i++) { await sleep(200); if (await evaluate(`!!document.querySelector('.zelda-library-ask')`)) break; }
step('library offers ask on the own document', await evaluate(`[...document.querySelectorAll('.zelda-library-ask')].map(b => ({ id: b.dataset.documentId, name: b.dataset.documentName, text: b.innerText.trim() }))`));
await shot('02_owner_library_ask_button.png');
await evaluate(`document.querySelector('.zelda-library-ask').click(), true`);
await sleep(300);
step('document chosen from the library', await state());
for (const q of ['How much has ManyChat raised?', 'How many users does ManyChat have?', 'How many messages does ManyChat send?', 'Should I invest in ManyChat?']) {
  step(`document question: ${q}`, await ask(q));
  if (q.includes('users')) await shot('03_owner_q2_traction_excerpt.png');
  if (q.includes('invest')) await shot('04_owner_q7_no_evidence.png');
}
await evaluate(`document.getElementById('zelda-ask-document-clear').click(), true`);
await sleep(300);
step('document cleared', await state());
step('marketplace question after clearing', await ask('Series A founders in messaging'));
await shot('05_owner_back_to_marketplace.png');

// Stranger: no own documents, so no ask button; forcing the context is refused by the server.
await signIn('verify_stranger');
await evaluate(`document.querySelector('.zelda-tab[data-tab="library"]').click(), true`);
await sleep(2500);
step('stranger library has no ask button', await evaluate(`({ askButtons: document.querySelectorAll('.zelda-library-ask').length, libraryText: document.getElementById('library-content').innerText.slice(0, 200) })`));
await evaluate(`setZeldaAskDocument(${seed.document_id}, 'forced'), true`);
step('stranger forcing another owner\'s document', await ask('How much has ManyChat raised?'));
await shot('06_stranger_forced_refused.png');
step('stranger marketplace question still works', await (async () => { await evaluate(`setZeldaAskDocument(null), true`); return ask('Series A founders in messaging'); })());

walk.console_errors = consoleErrors;
walk.extraction_calls = (() => { try { return JSON.parse(readFileSync(join(out, 'extraction_calls.json'), 'utf8')).length; } catch { return 0; } })();
writeFileSync(join(out, 'walk.json'), JSON.stringify(walk, null, 1));
ws.close(); chrome.kill(); await sleep(500);
try { rmSync(profile, { recursive: true, force: true }); } catch {}
console.log('console errors:', consoleErrors.length, 'extraction calls:', walk.extraction_calls);
