def load(rev):
    src = subprocess.run(['git', 'show', f'{rev}:zelda_api/basis_fidelity.py'], capture_output=True, text=True,
                         encoding='utf-8', check=True).stdout
    mod = importlib.util.module_from_spec(importlib.util.spec_from_loader('zelda_api.basis_fidelity', loader=None))
    mod.__package__ = 'zelda_api'
    exec(compile(src, f'{rev}:zelda_api/basis_fidelity.py', 'exec'), mod.__dict__)
    return mod


head = subprocess.run(['git', 'rev-parse', '--short', 'HEAD'], capture_output=True, text=True).stdout.strip()
dirty = subprocess.run(['git', 'diff', '--quiet', 'HEAD', '--', 'zelda_api/basis_fidelity.py']).returncode

            out, ch = replay({'s': t}, items, notes)
            res.append(('kept' if out['s'] == t else 'CHANGED:' + ','.join(ch.get('s', [])), out['s']))
        rows.append((gname, t, res))
sys.modules['zelda_api.basis_fidelity'], zelda_api.basis_fidelity = current, current
print('versions:', ' | '.join(v for v, _ in versions))
for gname, t, res in rows:
