"""Review of 9b28e14: the reviewer's two cited inputs, this round's controls and the earlier rounds' reviewer controls,
through the full normalisation -> time -> funding -> basis sequence on memo 101's payload, with the basis guard
loaded from git at 0664563, 7fb484b, 377a142 and 9b28e14 and from the working tree. No model call, no database."""
import importlib.util, subprocess, sys
import zelda_api
from zelda_api import basis_fidelity as current
from zelda_api.tests_output_fidelity import (replay, items_from, notes_from, memo101_payload, REVIEW_377_KEPT,
                                             REVIEW_377_REPAIRED, REVIEW_7FB_KEPT, REVIEW_750_KEPT, CONT_LEAD,
                                             REVIEW_9B2_KEPT, REVIEW_9B2_REPAIRED, MEMO98_JUDGMENT)
p = memo101_payload(); items, notes = items_from(p), notes_from(p)
GROUPS = [('reviewer 9b28e14 (must keep)', REVIEW_9B2_KEPT[:2]), ('control 9b28e14 (must keep)', REVIEW_9B2_KEPT[2:]),
          ('control 9b28e14 (must repair, expected output asserted in tests)', tuple(t for t, _, _ in REVIEW_9B2_REPAIRED)),
          ('memo 98 conditional (must repair)', (MEMO98_JUDGMENT,)),
          ('carried 377a142 (must keep)', REVIEW_377_KEPT[:2]), ('carried 377a142 controls (must keep)', REVIEW_377_KEPT[2:]),
          ('carried 377a142 (must repair)', REVIEW_377_REPAIRED + (CONT_LEAD + 'possibly overstated [S2, S3].',)),
          ('carried 7fb484b (must keep)', REVIEW_7FB_KEPT), ('carried 750de3f (must keep)', REVIEW_750_KEPT)]


def load(rev):
    src = subprocess.run(['git', 'show', f'{rev}:zelda_api/basis_fidelity.py'], capture_output=True, text=True,
                         encoding='utf-8', check=True).stdout
    mod = importlib.util.module_from_spec(importlib.util.spec_from_loader('zelda_api.basis_fidelity', loader=None))
    mod.__package__ = 'zelda_api'
    exec(compile(src, f'{rev}:zelda_api/basis_fidelity.py', 'exec'), mod.__dict__)
    return mod


head = subprocess.run(['git', 'rev-parse', '--short', 'HEAD'], capture_output=True, text=True).stdout.strip()
dirty = subprocess.run(['git', 'diff', '--quiet', 'HEAD', '--', 'zelda_api/basis_fidelity.py']).returncode
versions = [('0664563', load('0664563')), ('7fb484b', load('7fb484b')), ('377a142', load('377a142')), ('9b28e14', load('9b28e14')),
            (f'{head} (working tree{", DIRTY" if dirty else ", clean"})', current)]
rows = []
for gname, texts in GROUPS:
    for t in texts:
        res = []
        for label, mod in versions:
            sys.modules['zelda_api.basis_fidelity'], zelda_api.basis_fidelity = mod, mod
            out, ch = replay({'s': t}, items, notes)
            res.append(('kept' if out['s'] == t else 'CHANGED:' + ','.join(ch.get('s', [])), out['s']))
        rows.append((gname, t, res))
sys.modules['zelda_api.basis_fidelity'], zelda_api.basis_fidelity = current, current
print('versions:', ' | '.join(v for v, _ in versions))
for gname, t, res in rows:
    print(f'[{gname}] {t}')
    for (label, _), (verdict, out) in zip(versions, res):
        print(f'    {label:>32}: {verdict}' + ('' if verdict == 'kept' else f'\n{"":38}-> {out}'))
