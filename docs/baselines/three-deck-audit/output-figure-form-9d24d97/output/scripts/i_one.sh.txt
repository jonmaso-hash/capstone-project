#!/bin/sh
# Usage: i_one.sh <chunk number> -- the Nth chunk of 5 IDs from i_ids.txt, foreground, gated at 1.2 GB free.
O=/c/Users/jonathan/AppData/Local/Temp/claude/C--Users-jonathan-Desktop-KCV-np/c8a313d5-81e1-402a-b7da-9205b6f20fab/scratchpad/of705
ids=$(cut -d' ' -f2- $O/i_ids.txt | tr ' ' '\n' | sed -n "$(( ($1-1)*5+1 )),$(( $1*5 ))p" | tr '\n' ' ')
f=$(powershell -NoProfile -Command "[math]::Round((Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory/1MB,2)")
awk -v f="$f" 'BEGIN{exit !(f>=1.2)}' || { echo "STOP before chunk $1: free=$f" | tee -a $O/i_mutation_ram.txt; exit 3; }
sh $O/mut_chunk_i.sh c$(printf %02d $1) $ids
