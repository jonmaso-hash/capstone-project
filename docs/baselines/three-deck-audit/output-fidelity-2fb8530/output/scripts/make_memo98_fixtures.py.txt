"""Write zelda_api/data/funding_fidelity/memo98_payload.json and memo98.json from the committed memo-98 evidence
(0dd8e91): the grounded payload exactly as sent to the model, and the model's raw reply with the stored sections.
Normalised items and model output only -- no chunk text, key text or canary. Usage: python make_memo98_fixtures.py <worktree>"""
import json, os, subprocess, sys

WT = sys.argv[1]
EV = 'docs/baselines/manychat/usage-basis-d276918/pass-a'
show = lambda p: subprocess.run(['git', 'show', f'0dd8e91:{EV}/{p}'], capture_output=True, text=True, encoding='utf-8',
                                cwd=WT, check=True).stdout
call = json.loads(show('09b_model_call.json'))
user = call['request']['messages'][0]['content']
start = user.find('{', user.find('## EVIDENCE'))
payload, _ = json.JSONDecoder().raw_decode(user[start:])
memo = json.loads(show('09_memo.json'))['memo']
out = os.path.join(WT, 'zelda_api', 'data', 'funding_fidelity')
blob = json.dumps(payload) + call['response']['text'][0]
for leak in ('CANARY', 'angel.co', 'mike@manychat.com', 'ANSWER_KEY', 'ubmc'):
    assert leak not in blob, leak
with open(os.path.join(out, 'memo98_payload.json'), 'w', encoding='utf-8', newline='\n') as f:
    json.dump({'source': f'{EV}/09b_model_call.json at 0dd8e91 (memo 98, document 3140): the evidence payload as sent',
               'payload': payload}, f, indent=1, ensure_ascii=False)
    f.write('\n')
sections = {k: v for k, v in memo.items() if isinstance(v, str) and k not in ('created_at', 'updated_at')}
with open(os.path.join(out, 'memo98.json'), 'w', encoding='utf-8', newline='\n') as f:
    json.dump({'source': f'{EV}/09b_model_call.json and 09_memo.json at 0dd8e91 (memo 98, document 3140)',
               'model_reply_text': call['response']['text'][0], 'stored': sections}, f, indent=1, ensure_ascii=False)
    f.write('\n')
print('written', len(payload['items']), 'items;', len(sections), 'stored text fields; notes', payload.get('consistency_notes'))
