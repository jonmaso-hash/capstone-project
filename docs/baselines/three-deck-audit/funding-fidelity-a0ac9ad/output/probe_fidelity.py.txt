items = [NS(ref=i['ref'], kind=i['kind'], category=i['category'], statement=i['statement'], narrator=i.get('narrator', ''),

memo = json.load(open(EV + r'\09_memo.json', encoding='utf-8'))['memo']
for key, value in memo.items():
    if not isinstance(value, str):
        continue
    new, changes = guard_funding_fidelity(value, facts)
    if changes:
        print('\n==', key, changes)
        for a, b in zip(value.splitlines(), new.splitlines()):
            if a != b:
                print('  -', a)
                print('  +', b)
