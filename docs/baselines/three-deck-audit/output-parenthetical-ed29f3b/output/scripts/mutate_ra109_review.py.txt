def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()


results = []
for mid, what, rel, old, new in M:
    if ONLY and mid not in ONLY:
        continue
    path = os.path.join(WT, rel)
    original = open(path, 'rb').read()
    before = sha(path)
    text = original.decode('utf-8')
    if '\r\n' in text:
        old, new = old.replace('\n', '\r\n'), new.replace('\n', '\r\n')
    count = text.count(old)
    if count != 1:
        results.append((mid, what, f'NOT APPLIED (anchor found {count}x)'))
        print(mid, 'NOT APPLIED', count, flush=True)
        continue
    try:
        open(path, 'wb').write(text.replace(old, new).encode('utf-8'))
        run = subprocess.run(['sh', T, WT, *LABELS], capture_output=True, text=True, encoding='utf-8', errors='replace')
        out = run.stdout + run.stderr
        summary = [l for l in out.splitlines() if l.startswith(('Ran ', 'OK', 'FAILED'))]
        failing = sorted({l.split(' ')[1] for l in out.splitlines() if l.startswith(('FAIL:', 'ERROR:'))})
        caught = not any(l.startswith('OK') for l in summary)
        results.append((mid, what, ('CAUGHT' if caught else 'SURVIVED') + ' ' + ' '.join(summary), failing))
    finally:
        open(path, 'wb').write(original)
    restored = sha(path) == before
    print(mid, results[-1][2], 'restored' if restored else 'RESTORE FAILED', flush=True)
    if not restored:
        sys.exit('restore failed for ' + rel)

print()
for r in results:
    print(' | '.join(r[:3]))
    if len(r) > 3:
        print('    failing:', ', '.join(r[3][:8]) + (' ...' if len(r[3]) > 8 else ''))
