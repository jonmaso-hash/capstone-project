     'name a ref the payload supplies'),
    ("BASE = '066456341935d5ace32760a00788f78ec61a047a'", "BASE = 'd0bad8fdaace39f2f64be9f576874da9b0bf5e7a'"),
    ("'zelda_api.basis_fidelity_0664563'", "'zelda_api.basis_fidelity_d0bad8f'"),
    ("'0664563:zelda_api/basis_fidelity.py'", "'d0bad8f:zelda_api/basis_fidelity.py'"),
    ("    'memo101': ('82771ef', 'docs/baselines/manychat/output-fidelity-0664563/pass-a'),\n}",
     "    'memo101': ('82771ef', 'docs/baselines/manychat/output-fidelity-0664563/pass-a'),\n"
     "    'memo102': ('6d131f5', 'docs/baselines/manychat/output-fidelity-d0bad8f/pass-a'),\n}"),
    ("if name in ('memo100', 'memo101') else None,", "if name in ('memo100', 'memo101', 'memo102') else None,"),
    ("        'refs_in_order_same_as_base_replay':",
     "        'every_ref_supplied': all(set(refs(new[k])) <= {i.ref for i in items} for k in raw if isinstance(new[k], str)),\n"
     "        'refs_in_order_same_as_base_replay':"),
]

    s = s.replace(a, b)

print('derived replay.py')
