// RA-114 revision 2 probe (run tooling). Served by the run server at /__rev2/probe.js. Every request below
// is same-origin, as the account signed in by hand; the run server records it (rev2_runtime.RecorderMiddleware).
// Order and model-call allowances are enforced on the server (ordered checkpoints + the gate). Any safety
// failure here (wrong account, wrong document, a checkpoint refused, an unexpected answer shape) aborts the
// whole run on the server: no step retries anything.
'use strict';
const T = window.REV2_TOKEN;
let ST = null, BUSY = false;
const $ = id => document.getElementById(id);
const sleep = ms => new Promise(r => setTimeout(r, ms));
const log = (msg, cls) => { const d = document.createElement('div'); if (cls) d.className = cls; d.textContent = `${new Date().toISOString()} ${msg}`; $('log').prepend(d); };
const csrf = () => (document.cookie.match(/(?:^|; )csrftoken=([^;]+)/) || [])[1] || '';

async function rev2(path, body) {
  const r = await fetch(`/__rev2/${path}`, body === undefined ? { headers: { 'X-Rev2-Token': T }, cache: 'no-store' }
    : { method: 'POST', headers: { 'X-Rev2-Token': T, 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
  return r.json();
}
async function refresh() { ST = await rev2('state'); render(); return ST; }
async function abort(reason) { log('ABORT: ' + reason, 'abort'); await rev2('abort', { reason }); await refresh(); throw new Error('aborted: ' + reason); }
async function record(step, data) { await rev2('record', { step, data }); }
async function checkpoint(name) {
  const r = await rev2('checkpoint', { name });
  ST = r.state; render();
  if (!r.ok) { log(`checkpoint ${name} refused: ${r.problem}`, 'abort'); throw new Error('checkpoint refused'); }
  log(`checkpoint ${name}: ledger ${ST.ledger.lines} line(s), next phase ${ST.phase} (allows ${ST.allowance})`);
}
async function app(path, { method = 'GET', json, step, form, redirect = 'follow' } = {}) {
  const headers = { 'X-Rev2-Step': step || '' };
  if (method !== 'GET') headers['X-CSRFToken'] = csrf();
  if (json !== undefined) headers['Content-Type'] = 'application/json';
  const r = await fetch(path, { method, headers, credentials: 'same-origin', redirect, cache: 'no-store',
                                body: json !== undefined ? JSON.stringify(json) : form });
  let body = null;
  if (r.type !== 'opaqueredirect') { const t = await r.text(); try { body = JSON.parse(t); } catch { body = t; } }
  return { status: r.status, type: r.type, body };
}

// Identity: the server's view of the session AND the nav text of a real page (revision 1/base: check the nav
// before counting a check; Chrome autofill signed in the wrong account last time).
async function requireAccount(role) {
  await refresh();
  const who = ST.whoami;
  if (role === 'ANON') { if (who.authenticated) await abort(`expected signed out, found ${who.username}`); return; }
  const want = ST.accounts[role.toLowerCase()];
  if (!who.authenticated || who.username !== want) await abort(`expected ${want}, signed in as ${who.authenticated ? who.username : 'nobody'}`);
  const page = await app('/home/', { step: 'identity:' + role });
  const nav = typeof page.body === 'string' ? page.body : '';
  const text = new DOMParser().parseFromString(nav, 'text/html').body.textContent || '';
  if (!text.includes(want)) await abort(`nav text does not show ${want}`);
  log(`identity ${role}: ${want} (server session and nav text)`);
}
function requirePhase(...phases) { if (!phases.includes(ST.phase)) throw new Error(`this step needs phase ${phases.join('/')}, run is in ${ST.phase}`); }
const fill = (tpl, ids) => tpl.replace(/\{(\w+)\}/g, (_, k) => { if (ids[k] == null) throw new Error('missing id ' + k); return ids[k]; });

async function matrix(label) {
  const ids = ST.ids, rows = [];
  for (const [surface, tpl] of ST.matrix) {
    const r = await app(fill(tpl, ids), { step: `matrix:${label}:${surface}`, redirect: 'manual' });
    rows.push([surface, r.status, r.type]);
  }
  for (const [surface, path] of [['M_search', 'search'], ['N_rag', 'rag']]) {
    const r = await app(`/api/v1/zelda/documents/${ids.doc}/${path}/`, { method: 'POST', json: { query: ST.questions.Q1, top_k: 5 }, step: `matrix:${label}:${surface}` });
    rows.push([surface, r.status]);
  }
  await record('matrix:' + label, { rows });
  log(`matrix ${label}: ${rows.length} rows`);
}
async function authProbe(label) {
  const doc = ST.doc_id;
  const ask = await app('/api/v1/zelda/ask/', { method: 'POST', json: { q: ST.questions.Q1, document_id: doc }, step: `auth:${label}:ask` });
  const search = await app(`/api/v1/zelda/documents/${doc}/search/`, { method: 'POST', json: { query: ST.questions.Q1, top_k: 5 }, step: `auth:${label}:search` });
  await record('auth:' + label, { ask_status: ask.status, search_status: search.status });
  log(`authorization ${label}: ask ${ask.status}, search ${search.status} (expected 403, 403)`, ask.status === 403 && search.status === 403 ? '' : 'abort');
}

async function panelCheck() {
  requirePhase('controls');
  await checkpoint('before_panel');
  const doc = ST.doc_id, Q1 = ST.questions.Q1, obs = { document_id: doc };
  const w = window.open('/home/', 'rev2panel');
  if (!w) await abort('panel: the popup was blocked (allow popups for 127.0.0.1:8000 before the run; the check is not repeated)');
  const until = async (fn, what, ms = 30000) => { for (let t = 0; t < ms; t += 250) { try { const v = fn(); if (v) return v; } catch {} await sleep(250); } await abort('panel: timed out waiting for ' + what); };
  await until(() => w.document.readyState === 'complete' && w.document.getElementById('zelda-ask-form'), 'the panel');
  w.document.querySelector('.zelda-tab[data-tab="library"]').click();
  const button = await until(() => w.document.querySelector(`.zelda-library-ask[data-document-id="${doc}"]`), `"Ask about this document" for document ${doc}`);
  obs.button_document_name = button.dataset.documentName;
  button.click();
  const bar = w.document.getElementById('zelda-ask-document');
  obs.context_bar_shown = bar.style.display === 'block';
  obs.context_bar_name = w.document.getElementById('zelda-ask-document-name').textContent;
  if (!obs.context_bar_shown) await abort('panel: context bar not shown');
  const before = ((await refresh()).ask_posts_by_phase.panel || 0);
  w.document.getElementById('zeldaAskQuery').value = Q1;
  w.document.getElementById('zelda-ask-form').requestSubmit();
  for (let t = 0; t < 30000; t += 500) { await sleep(500); const s = await refresh(); if ((s.ask_posts_by_phase.panel || 0) > before) break; }
  await sleep(1500);
  const list = w.document.getElementById('ask-results-list');
  const cards = [...list.children];
  obs.rendered = cards.map(c => ({ label: (c.querySelector('.small') || {}).textContent || null, has_quote: !!c.querySelector('blockquote'),
                                   quote_head: ((c.querySelector('blockquote') || {}).textContent || '').slice(0, 80) }));
  obs.marketplace_cards = cards.filter(c => !c.querySelector('blockquote')).length;
  obs.results_card_shown = w.document.getElementById('ask-results-card').style.display === 'block';
  obs.ask_posts_in_panel_after_q1 = (await refresh()).ask_posts_by_phase.panel || 0;
  w.document.getElementById('zelda-ask-document-clear').click();
  obs.context_bar_hidden_after_clear = bar.style.display === 'none';
  await sleep(3000);
  obs.ask_posts_in_panel_after_clear = (await refresh()).ask_posts_by_phase.panel || 0;
  w.close();
  await record('panel', obs);
  log('panel: ' + JSON.stringify(obs).slice(0, 400));
  if (obs.ask_posts_in_panel_after_q1 !== 1 || obs.ask_posts_in_panel_after_clear !== 1) await abort(`panel sent ${obs.ask_posts_in_panel_after_clear} /ask/ request(s); exactly 1 expected`);
  await checkpoint('after_panel');
}

function render() {
  const s = ST;
  $('state').textContent = s.error ? s.error : [
    `mode: ${s.mode}${s.abort ? '   ABORTED: ' + s.abort.reason : ''}`,
    `signed in: ${s.whoami.authenticated ? s.whoami.username + (s.whoami.is_staff ? ' (staff, MFA stamp ' + s.whoami.staff_mfa_stamp_present + ')' : '') : 'nobody'}`,
    `phase: ${s.phase} (allows ${s.allowance}, used ${s.used})   run total: ${s.total} of 8   next checkpoint: ${s.next_checkpoint}`,
    `ledger: ${s.ledger.lines} line(s) sha256 ${s.ledger.sha256.slice(0, 16)}…   document: ${s.doc_id ?? '—'}`].join('\n');
  const box = $('steps');
  if (!box.children.length) STEPS.forEach(st => { const b = document.createElement('button'); b.id = 'step-' + st.id; b.textContent = st.label; b.onclick = () => go(st); box.append(b); });
  STEPS.forEach(st => { $('step-' + st.id).disabled = BUSY || !!s.abort; });
}
async function go(step) {
  if (BUSY) return;
  BUSY = true; render();
  try {
    if (step.role) await requireAccount(step.role);
    log(`step ${step.id} started`);
    await step.run();
    await record('step_done', { step: step.id });
    log(`step ${step.id} done`);
  } catch (e) { log(`step ${step.id} stopped: ${e.message}`, 'abort'); await record('step_stopped', { step: step.id, error: String(e.message) }).catch(() => {}); }
  BUSY = false; await refresh();
}
window.REV2 = { go: id => go(STEPS.find(s => s.id === id)), refresh, state: () => ST };
refresh();
