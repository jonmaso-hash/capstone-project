"""Targeted mutations for F-1 / F-2 (memo 104): each removes the repair (the memo-104 defect returns) or a protection of
the new rule (harmful overcorrection). Same harness as the earlier rounds: one mutant at a time on the worktree's
guard module, the affected test modules run against it, the original bytes restored and SHA-256 checked before the
next. A mutant counts as CAUGHT only by assertion failures; errors alone are CRASH-ONLY; no test run is NO-RUN.
Usage: python mutate_f12.py <worktree> [ID ...]"""
import hashlib, os, re, subprocess, sys

WT = sys.argv[1]
ONLY = set(sys.argv[2:])
FILES = {'basis': os.path.join(WT, 'zelda_api', 'basis_fidelity.py'),
         'funding': os.path.join(WT, 'zelda_api', 'funding_fidelity.py')}
T = os.path.join(os.path.dirname(os.path.abspath(__file__)), 't.sh')
LABELS = ['zelda_api.tests_output_fidelity', 'zelda_api.tests_funding_fidelity']
M = [
    # --- F-1: whole-word figure tail
    ('F1', 'funding', 'F-1 returns: the tail alternatives end anywhere (5120190 pattern)',
     r"(?:funding|raised|figure|total|listed|reported)\b)*')", r"(?:funding|raised|figure|total|listed|reported))*')"),
    ('F2', 'funding', 'partial fix: only "reported" is bounded; the other five alternatives still end inside a word',
     r"(?:funding|raised|figure|total|listed|reported)\b)*')", r"(?:funding|raised|figure|total|listed|reported\b))*')"),
    # --- F-2: heading scope
    ('H1', 'basis', 'F-2 returns: the heading rule is not applied',
     "        sentence = _heading_scope(sentence, facts, changes)\n", ""),
    ('H2', 'basis', 'overcorrection: any subject (no own source required; a narrator or analyst is restated as the deck\'s)',
     "        if (not _OWN_SOURCE.search(subject) or any(name.lower() in subject.lower() for name in facts.narrators)\n"
     "                or _ATTRIBUTION.search(subject) or set(words) & _DENIAL or _UNCERTAIN.search(' '.join(words))):",
     "        if (set(words) & _DENIAL or _UNCERTAIN.search(' '.join(words))):"),
    ('H3', 'basis', 'overcorrection: a hedged or denied subject is restated',
     " or set(words) & _DENIAL or _UNCERTAIN.search(' '.join(words))):\n            continue                           # only the company's",
     "):\n            continue                           # only the company's"),
    ('H4', 'basis', 'overcorrection: a heading phrase inside a condition is acted on',
     "        if sentence[:at].count('(') != sentence[:at].count(')') or _conditional(sentence, at):",
     "        if sentence[:at].count('(') != sentence[:at].count(')'):"),
    ('H5', 'basis', 'overcorrection: a mid-clause heading phrase is acted on',
     "        if not _CLAUSE_END.match(sentence, end):\n            continue                           # mid-clause",
     "        if False:\n            continue                           # mid-clause"),
    ('H6', 'basis', 'overcorrection: the list may hold other facts, figures, intervals or asides',
     "        if not _placing_words(listed, vocabulary, False):", "        if False:"),
    ('H7', 'basis', 'overcorrection: a list none of whose refs carries the heading is restated',
     "        if not outside or len(outside) == len(cited):", "        if not outside:"),
    ('H8', 'basis', 'overcorrection: refs outside the pair (another fact) may move',
     "        both = next((pair for pair in facts.pairs if len(pair.sides) == 2 and set(outside) <= pair.members",
     "        both = next((pair for pair in facts.pairs if len(pair.sides) == 2"),
    ('H9', 'basis', 'wrong output: the memo\'s pointer to the note is dropped for the clarification',
     "        body += ''.join(' ' + pointer for pointer in pointers) or f' ({both", "        body += '' or f' ({both"),
    ('H10', 'basis', 'wrong output: the list\'s closing dash is not restored before the next clause',
     "        if dash and re.match(r'\\s*,', rest):", "        if False:"),
    ('H11', 'basis', 'wrong scope: every cited ref is placed under the heading (page 12 included)',
     "        outside = [ref for ref in cited if places[ref][1] != claimed]\n        if not outside",
     "        outside = [ref for ref in cited if places[ref][1] != claimed]\n        places = {r: (p, claimed, f) for r, (p, _c, f) in places.items()}\n        if not outside"),
]


def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()


originals = {k: open(p, 'rb').read() for k, p in FILES.items()}
before = {k: sha(p) for k, p in FILES.items()}
for mid, which, desc, old, new in M:
    if ONLY and mid not in ONLY:
        continue
    path, original = FILES[which], originals[which]
    text = original.decode('utf-8')
    if '\r\n' in text:                         # the working copy's line endings (core.autocrlf)
        old, new = old.replace('\n', '\r\n'), new.replace('\n', '\r\n')
    count = text.count(old)
    if count != 1:
        print(f'{mid} NOT-APPLIED anchor count {count} | {desc}', flush=True)
        continue
    open(path, 'wb').write(text.replace(old, new).encode('utf-8'))
    try:
        run = subprocess.run(['sh', T, WT] + LABELS, capture_output=True, text=True, encoding='utf-8', errors='replace')
        out = run.stdout + run.stderr
    finally:
        open(path, 'wb').write(original)
    restored = all(sha(p) == before[k] for k, p in FILES.items())
    summary = (re.findall(r'^(?:OK.*|FAILED \(.*\))$', out, re.M) or ['?'])[-1]
    ran = (re.findall(r'^Ran \d+ tests.*$', out, re.M) or ['?'])[-1]
    fails = sorted({m for m in re.findall(r'^FAIL: (\w+)', out, re.M)})
    errors = sorted({m for m in re.findall(r'^ERROR: (\w+)', out, re.M)})
    verdict = 'CAUGHT' if fails else ('CRASH-ONLY' if errors else 'SURVIVED')
    if ran == '?':
        verdict = 'NO-RUN'
    print(f'{mid} {verdict} {ran} {summary} restored={restored} | {desc}', flush=True)
    print(f'    assertion failures in: {", ".join(fails) or "-"}', flush=True)
    print(f'    errors in: {", ".join(errors) or "-"}', flush=True)
    for case in sorted(set(re.findall(r'^FAIL: \w+ \([^)]*\) \((.*)\)$', out, re.M)))[:6]:
        print(f'    failing sub-case: {case[:200]}', flush=True)
    if not restored:
        sys.exit('STOP: restore hash mismatch after ' + mid)
print('final sha matches pre-run:', all(sha(p) == before[k] for k, p in FILES.items()), {k: sha(p) for k, p in FILES.items()})
