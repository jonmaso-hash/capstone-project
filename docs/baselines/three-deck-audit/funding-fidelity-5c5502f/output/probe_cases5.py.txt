"""The three remaining cases at 304dede, before (304dede's funding_fidelity.py and claim_context.py) and after
(worktree), in memory, no DB, in the pipeline's order: time guard, then funding guard. Each output is guarded again
(idempotence) and every case also runs in reversed item order.
Usage: python probe_cases5.py <worktree-after> <dir with funding_fidelity_304dede.py and claim_context_304dede.py>"""
import importlib.util, os, sys
from types import SimpleNamespace as NS

AFTER, BEFORE = sys.argv[1], sys.argv[2]
sys.path.insert(0, AFTER)
import zelda_api.claim_context as cc_after
import zelda_api.funding_fidelity as ff_after


def load(name, filename):
    spec = importlib.util.spec_from_file_location(f'zelda_api.{name}', os.path.join(BEFORE, filename))
    module = importlib.util.module_from_spec(spec)
    module.__package__ = 'zelda_api'
    spec.loader.exec_module(module)
    return module


cc_before, ff_before = load('cc_before', 'claim_context_304dede.py'), load('ff_before', 'funding_fidelity_304dede.py')


def item(ref, statement, category='Financials', narrator='', as_of='', kind='statement'):
    return NS(ref=ref, kind=kind, category=category, statement=statement, narrator=narrator, as_of=as_of,
              sources=(NS(profile_field=''),), dated=True, time_wording_allowed=())


BASE = [item('C1', '$23.1M raised', 'funding_raised', 'Pitch Deck Hunt', 'not stated', 'claim'),
        item('S1', '140 000+ bots', 'Traction', as_of='not stated'),
        item('S3', '$23.1M raised', 'Funding', 'Pitch Deck Hunt', 'not stated'),
        item('S4', '$22M raised', 'Funding', 'Crunchbase', 'not stated'),
        item('S5', '$20M raised', 'Funding', 'Dealroom', 'not stated')]
PAIR = [item('S6', '$2.1M cash in the bank', as_of='2016-04-30'),
        item('S7', 'Net burn of $150K per month', as_of='2016-04-30')]
TWO_CASH = [item('S6', '$2.1M cash in the bank', as_of='2016-04-01'),
            item('S8', '$1.9M cash in the bank', as_of='2016-04-30'),
            item('S7', 'Net burn of $150K per month', as_of='2016-04-30')]
CASES = [
    ('1. "Today," with cash and monthly burn both dated 2016-04-30',
     'Today, the $23.1M funding figure (per Pitch Deck Hunt) [S3] supports remaining runway.', BASE + PAIR),
    ('1. "As of today," (the time guard removes it first)',
     'As of today, the $23.1M funding figure (per Pitch Deck Hunt) [S3] supports remaining runway.', BASE + PAIR),
    ('1. control: "Now that ..." is not a time qualifier',
     'Now that funds are raised, the $23.1M funding figure (per Pitch Deck Hunt) [S3] supports remaining runway.',
     BASE + PAIR),
    ('2. one span, [S1, S3] with [S4] plus [S5]',
     'The self-reported figures [S1, S3] with [S4] plus [S5] are unverified.', BASE),
    ('2. funding [S3] plus [S4]', 'The self-reported funding [S3] plus [S4] is unverified.', BASE),
    ('3. cash dated 2016-04-01 and 2016-04-30, current premise',
     'It would anchor the $23.1M funding figure (per Pitch Deck Hunt) [S3] against current remaining runway.',
     BASE + TWO_CASH),
]


def run(cc, ff, text, items):
    allowed = {i.ref: i.time_wording_allowed for i in items if i.dated}
    timed = cc.guard_time_wording(text, allowed)[0]
    facts = ff.funding_facts(items, 'ManyChat')
    once = ff.guard_funding_fidelity(timed, facts)[0]
    again = ff.guard_funding_fidelity(cc.guard_time_wording(once, allowed)[0], facts)[0]
    return once, once == again


for name, text, items in CASES:
    print('==', name)
    print('   in       :', text)
    for label, cc, ff in (('304dede', cc_before, ff_before), ('after', cc_after, ff_after)):
        for order, its in (('', items), (' reversed', list(reversed(items)))):
            once, idem = run(cc, ff, text, its)
            print(f'   {label + order:17}:', once, '' if idem else '   [NOT IDEMPOTENT]')
    print()
