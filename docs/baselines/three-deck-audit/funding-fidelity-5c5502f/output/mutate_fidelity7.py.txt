"""Mutations for the three remaining cases at 304dede. Usage: python mutate_fidelity7.py <worktree> [ids...|--check|--ids]
Inherits the 85 mutations of mutate_fidelity6.py (targets unchanged; checked with --check). O: this change.
Tests run: zelda_api.tests_funding_fidelity and zelda_api.tests_claim_context_dates (the time guard changed too).
One mutation at a time, file restored in `finally`; never wrap this in `timeout`."""
import os, re, subprocess, sys
HERE = os.path.dirname(os.path.abspath(__file__))
WT = sys.argv[1]
p6 = os.path.join(HERE, 'mutate_fidelity6.py')
ns = {'__file__': p6}
exec(open(p6, encoding='utf-8').read().split(chr(10) + 'env = dict(')[0], ns)
FF = ns['FF']
CC = 'zelda_api/claim_context.py'
RETARGET = {   # the introductory-qualifier code sits between these targets now; same intent
    'J13': ("    if not (_cited(sentence) & facts.funding_refs or any(p.search(sentence) for p in facts.funding_figures)):\n"
            "        return sentence\n\n    lead = _LEAD_NOW", "\n    lead = _LEAD_NOW"),
    'M1': ("current=bool(match.group('now') or _NOW.search(clause) or lead))", "current=False)"),
    'M2': ("current=bool(match.group('now') or _NOW.search(clause) or lead))", "current=bool(match.group('now') or lead))"),
}
MUTATIONS = [(n, r, *RETARGET.get(n.split()[0], (o, w))) for n, r, o, w in ns['MUTATIONS']] + [
    # 1. An introductory time qualifier governs the premise
    ('O1 introductory qualifier ignored', FF,
     "_NOW.search(clause) or lead))", "_NOW.search(clause)))"),
    ('O2 introductory qualifier kept beside the dated note', FF,
     "    if lead and NO_CURRENT in rewritten and NO_CURRENT not in sentence:\n", "    if False:\n"),
    ('O3 qualifier removed without recapitalising', FF,
     "        rewritten = lead.group('lead') + rest[:1].upper() + rest[1:]\n",
     "        rewritten = lead.group('lead') + rest\n"),
    ('O4 "Now that ..." read as a time qualifier', FF,
     r"|now\s*,\s*)', re.I)", r"|now\b\s*,?\s*)', re.I)"),
    ('O9 time guard: opening not restored', CC,
     "                rewritten = _restore_opening(sentence, rewritten)\n", ""),
    ('O10 time guard: leading space kept', CC,
     "    body = rewritten[len(lead):].lstrip()\n", "    body = rewritten[len(lead):]\n"),
    ('O11 time guard: list marker not kept as the lead', CC,
     "    lead = original[:len(original) - len(original.lstrip())] + (marker.group(0) if marker else '')\n",
     "    lead = ''\n"),
    # 2. Every narrated group governed by one span
    ('O5 joined groups not merged', FF,
     "    merged = _merge_joined_groups(", "    merged = None and _merge_joined_groups("),
    ('O6 any text joins two groups', FF,
     "_JOINER.fullmatch(sentence, groups[j].end(), groups[j + 1].start())", "True"),
    ('O7 span not relocated after the merge', FF,
     "        span = _SELF_REPORTED.match(sentence, span.start() if span.start() < run_start else span.start() - removed)\n",
     "        span = span\n"),
    ('O12 company-only runs merged too', FF,
     "        if len(run) > 1 and any(movable(r) for r in refs):\n", "        if len(run) > 1:\n"),
    # 3. The latest ISO date uses its day
    ('O8 latest date ignores the day', FF,
     "        return ((n - 1) // 12, 1, as_of) if kind == 'month'", "        return ((n - 1) // 12, 1, n) if kind == 'month'"),
]
env = dict(os.environ, SECRET_KEY='ci-only-placeholder-key-not-for-production-use', DEBUG='True',
           ALLOWED_HOSTS='localhost,127.0.0.1', EMAIL_HOST_USER='ci@example.invalid', PYTHONIOENCODING='utf-8',
           CELERY_TASK_ALWAYS_EAGER='True', CELERY_BROKER_URL='memory://', CELERY_RESULT_BACKEND='cache+memory://')
only = sys.argv[2:] or None
if only == ['--check']:
    for name, rel, old, new in MUTATIONS:
        c = open(os.path.join(WT, rel), encoding='utf-8').read().count(old)
        if c != 1:
            print('TARGET', c, name)
    print(len(MUTATIONS), 'mutations checked')
    sys.exit(0)
if only == ['--ids']:
    print(' '.join(m[0].split()[0] for m in MUTATIONS))
    sys.exit(0)
for name, rel, old, new in MUTATIONS:
    if only and name.split()[0] not in only:
        continue
    path = os.path.join(WT, rel)
    original = open(path, encoding='utf-8', newline='').read()
    nl = '\r\n' if '\r\n' in original else '\n'
    o, n = old.replace('\n', nl), new.replace('\n', nl)
    if original.count(o) != 1:
        print('== ' + name + '\nHARNESS: target found %d times' % original.count(o), flush=True)
        continue
    try:
        open(path, 'w', encoding='utf-8', newline='').write(original.replace(o, n))
        out = subprocess.run([sys.executable, 'manage.py', 'test', '--noinput', '--parallel', '1',
                              'zelda_api.tests_funding_fidelity', 'zelda_api.tests_claim_context_dates'],
                             cwd=WT, env=env, capture_output=True, text=True, encoding='utf-8', errors='replace')
        lines = [re.sub(r' \(zelda_api\.tests_\w+\.\w+\.\w+\)', '', l)
                 for l in (out.stdout + out.stderr).splitlines() if re.match(r'^(FAIL|ERROR):|^Ran |^OK$|^FAILED', l)]
        print('==', name)
        print('\n'.join(lines), flush=True)
    finally:
        open(path, 'w', encoding='utf-8', newline='').write(original)
