figures, every equation in a sentence, shared company credit)
and FF-411 (Markdown charset) on fix/narrator-provenance.

One mutation at a time: the target file is rewritten with one exact substring replaced (asserted to occur
exactly once), the named test labels run in a fresh process (CI env, --parallel 1, no bytecode written), and
the original bytes are restored in `finally`. RED = the tests failed (mutation killed). Never wrap this in
`timeout`: a kill would skip the restore. Usage: python mutate_np.py <worktree> <ids...> (or "all")."""

ENV = dict(os.environ, SECRET_KEY='ci-only-placeholder-key-not-for-production-use', DEBUG='True',
           ALLOWED_HOSTS='localhost,127.0.0.1', EMAIL_HOST_USER='ci@example.invalid', CELERY_TASK_ALWAYS_EAGER='True',
           CELERY_BROKER_URL='memory://', CELERY_RESULT_BACKEND='cache+memory://', PYTHONIOENCODING='utf-8',
           PYTHONDONTWRITEBYTECODE='1')


def sha(path):
    return hashlib.sha256(open(os.path.join(WT, path), 'rb').read()).hexdigest()


ids = list(MUTATIONS) if sys.argv[2:] == ['all'] else sys.argv[2:]
before = {p: sha(p) for p in (FF, VIEWS)}
for mid in ids:
    what, path, old, new, labels = MUTATIONS[mid]
    full = os.path.join(WT, path)
    original = open(full, 'rb').read()
    text = original.decode('utf-8')
    assert text.count(old) == 1, (mid, text.count(old))
    try:
        open(full, 'wb').write(text.replace(old, new).encode('utf-8'))
        run = subprocess.run([sys.executable, 'manage.py', 'test', *labels, '--parallel', '1'], cwd=WT, env=ENV,
                             capture_output=True, text=True, encoding='utf-8', errors='replace')
    finally:
        open(full, 'wb').write(original)
    tail = [l for l in run.stderr.splitlines() if l.startswith(('Ran ', 'OK', 'FAILED'))]
    verdict = 'RED (killed)' if run.returncode != 0 else 'SURVIVED'
    print(f'{mid:4} {verdict:13} {what} | {" ".join(tail)}', flush=True)
after = {p: sha(p) for p in (FF, VIEWS)}
print('fingerprints unchanged:', before == after, {p: h[:16] for p, h in after.items()})
