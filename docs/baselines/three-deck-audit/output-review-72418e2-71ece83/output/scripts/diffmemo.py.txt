import json, sys
a = json.load(open(sys.argv[1], encoding='utf-8'))[sys.argv[3]]
b = json.load(open(sys.argv[2], encoding='utf-8'))[sys.argv[3]]


def walk(x, y, path):
    if isinstance(x, dict) and isinstance(y, dict):
        for k in sorted(set(x) | set(y)):
            walk(x.get(k), y.get(k), f'{path}.{k}')
    elif x != y:
        print(path)
        print('  NEW:', json.dumps(x, ensure_ascii=False)[:1500])
        print('  OLD:', json.dumps(y, ensure_ascii=False)[:1500])


walk(a, b, sys.argv[3])
