"""Stage 2 (after the memo): derive this run's HTTP probe, its config, the Markdown check, the replay analysis, the
sentence scan and the comparison helpers from the memo-103 run's (rc6c/). probe.py and scan's checks are copied
unchanged. probe_cfg.json: this run's outdir, document 3146, Entity Integrity report 45, principals m512mc_*; markers
and the seven questions unchanged. md_check.py: principal and document id only. analyze: paths, label, fixture prefix;
this round's review IDs added to the absent-from-input list. scan: header relabelled to memo 104. cmp_matrix: labels
and the previous run's names/ids (memo 103). Each replacement must match. Usage: python make_scripts2.py"""
import json, os, shutil

HERE = os.path.dirname(os.path.abspath(__file__))
BS = chr(92)
PREV = os.path.join(os.path.dirname(HERE), 'rc6c')


def derive(name, reps, target=None):
    s = open(os.path.join(PREV, name), encoding='utf-8').read()
    for a, b in reps:
        assert s.count(a) >= 1, (name, a)
        s = s.replace(a, b)
    open(os.path.join(HERE, target or name), 'w', encoding='utf-8', newline='\n').write(s)
    print('derived', target or name)


shutil.copy(os.path.join(PREV, 'probe.py'), os.path.join(HERE, 'probe.py'))

                         'output-fidelity-5120190', 'pass-a'])

cfg['owner'] = 'm512mc_owner'
cfg['principals'] = {'OWNER': 'm512mc_owner', 'LITE': 'm512mc_lite', 'PAID': 'm512mc_paid', 'STAFF': 'm512mc_staff', 'ANON': None}
