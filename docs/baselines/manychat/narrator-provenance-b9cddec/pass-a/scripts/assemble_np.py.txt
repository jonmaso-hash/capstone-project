Adds git checks that frozen artifacts are unchanged against their known commits."""

def read(name):
    raw = open(os.path.join(HERE, name), 'rb').read()
    if raw[:2] in (b'\xff\xfe', b'\xfe\xff'):
        return raw.decode('utf-16')
    return raw.decode('utf-8', errors='replace')


def git(*a):
    return subprocess.run(['git', *a], capture_output=True, text=True, cwd=WT).stdout.strip()


CHECK = r'^(CODE-CHECK|CODE-DIFF|BASELINE-ENV-CHECK|BROKER-CHECK|DB-CHECK|SCHEMA-CHECK|EAGER-CHECK|MODEL-CALL-LOG|WORKER-IDENTITY|REFUSING|CHECK OK|preflight ok|broker|result backend|cache|migrations)'
lines = lambda name: [re.sub(r"tracked_changes=\d+ \[.*\]$", 'tracked_changes=<__pycache__ only; see CODE-DIFF>', l)
                      for l in read(name).splitlines() if re.match(CHECK, l)]
snap = lambda name: [l for l in read(name).replace('\r', '').splitlines() if not l.startswith('Invalid line')]
hb, ha = json.loads(read('hashes_before.json')), json.loads(read('hashes_after.json'))

             'docs/baselines/nike-fy2026/PROTOCOL.md'):
    frozen[path] = {'blob_at_b9cddec': git('rev-parse', 'b9cddec:' + path), 'blob_in_worktree': git('hash-object', path),
                    'last_commit_touching': git('log', '-1', '--format=%h %ad %s', '--date=short', 'b9cddec', '--', path)}

}

                    'identical': bool(a) and a == b, 'ref_tip': git('rev-parse', '--short', ref)}
