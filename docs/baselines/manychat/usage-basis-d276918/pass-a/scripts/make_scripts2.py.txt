import os

HERE = os.path.dirname(os.path.abspath(__file__))
PREV = os.path.join(os.path.dirname(HERE), 'np')
NP_SP = r'C--Users-jonathan-Desktop-KCV-np\c8a313d5-81e1-402a-b7da-9205b6f20fab\scratchpad\np'
UB_SP = r'C--Users-jonathan-Desktop-KCV-np\c8a313d5-81e1-402a-b7da-9205b6f20fab\scratchpad\ub'


def derive(name, reps, target):
    s = open(os.path.join(PREV, name), encoding='utf-8').read()
    for a, b in reps:
        assert s.count(a) >= 1, (name, a)
        s = s.replace(a, b)
    open(os.path.join(HERE, target), 'w', encoding='utf-8', newline='\n').write(s)
    print('derived', target)
